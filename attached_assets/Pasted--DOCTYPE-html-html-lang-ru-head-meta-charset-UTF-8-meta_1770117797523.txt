<!DOCTYPE html>
<html lang="ru">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Игра для ❤️</title>
    <style>
        * { margin: 0; padding: 0; }
        body, html { 
            height: 100%; 
            overflow: hidden;
            background: linear-gradient(to bottom, #87CEEB, #98FB98);
            font-family: Arial, sans-serif;
        }
        #gameCanvas {
            display: block;
            width: 100vw;
            height: 100vh;
            cursor: pointer;
            touch-action: none;
        }
    </style>
</head>
<body>
    <canvas id="gameCanvas"></canvas>
    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        let centerX, centerY;
        const ringRadius = 320;
        const ball = { x: 0, y: 0, vx: 0, vy: 0, radius: 25, color: '#ff69b4' };
        let ropes = [];
        let activeRope = null;
        const gravity = 3.5;
        const damping = 0.95;  // ✅ УСИЛИЛИ ТОРМОЖЕНИЕ!
        let gameStarted = false;

        function spawnBall() {
            centerX = canvas.width * 0.5;
            centerY = canvas.height * 0.4;
            ball.x = centerX;
            ball.y = centerY;
            ball.vx = 0;
            ball.vy = 0;
            ropes = [];
            activeRope = null;
        }

        function constrainToRing() {
            const dx = ball.x - centerX;
            const dy = ball.y - centerY;
            const distance = Math.sqrt(dx*dx + dy*dy);
            
            if (distance + ball.radius > ringRadius && gameStarted) {
                const angle = Math.atan2(dy, dx);
                const contactX = centerX + Math.cos(angle) * ringRadius;
                const contactY = centerY + Math.sin(angle) * ringRadius;
                
                ball.x = centerX + Math.cos(angle) * (ringRadius - ball.radius);
                ball.y = centerY + Math.sin(angle) * (ringRadius - ball.radius);
                
                // ✅ СБАЛАНСИРОВАННЫЙ ТОЛЧОК
                const speed = Math.sqrt(ball.vx*ball.vx + ball.vy*ball.vy);
                ball.vx = -Math.cos(angle) * speed * 2;  // Было 3.9 → 2.2
                ball.vy = -Math.sin(angle) * speed * 2;
                
                if (ropes.length < 3) {
                    ropes.push({
                        targetX: contactX,
                        targetY: contactY,
                        life: 150
                    });
                }
            }
        }

        function updateRopes() {
            if (activeRope) {
                const dx = activeRope.targetX - ball.x;
                const dy = activeRope.targetY - ball.y;
                const dist = Math.sqrt(dx*dx + dy*dy);
                if (dist > 160) {  // ✅ ЕЩЁ БЛИЖЕ!
                    const force = (dist - 160) * 0.0012;  // ✅ ЕЩЁ СИЛЬНЕЕ!
                    ball.vx += (dx / dist) * force;
                    ball.vy += (dy / dist) * force;
                }
            }

            for (let i = 0; i < ropes.length; i++) {
                const rope = ropes[i];
                rope.life--;
                
                const dx = rope.targetX - ball.x;
                const dy = rope.targetY - ball.y;
                const dist = Math.sqrt(dx*dx + dy*dy);
                
                if (dist > 160) {  // Было 180 → 160
                    const force = (dist - 160) * 0.0012;  // Было 0.0008 → 0.0012
                    ball.vx += (dx / dist) * force;
                    ball.vy += (dy / dist) * force;
                }
                
                if (rope.life <= 0) {
                    ropes.splice(i, 1);
                    i--;
                }
            }
        }

        function getEventPos(e) {
            const rect = canvas.getBoundingClientRect();
            return {
                x: (e.clientX || e.touches[0].clientX) - rect.left,
                y: (e.clientY || e.touches[0].clientY) - rect.top
            };
        }

        // ✅ СТАРТОВЫЙ ТОЛЧОК СБАЛАНСИРОВАН
        canvas.addEventListener('click', (e) => {
            const pos = getEventPos(e);
            const dx = pos.x - ball.x;
            const dy = pos.y - ball.y;
            if (Math.sqrt(dx*dx + dy*dy) < ball.radius * 1.5) {
                if (!gameStarted) {
                    gameStarted = true;
                    const randomAngle = (Math.random() - 0.5) * Math.PI;
                    const startSpeed = 16;  // Было 27 → 16
                    ball.vx = Math.cos(randomAngle) * startSpeed;
                    ball.vy = Math.sin(randomAngle) * startSpeed + 5;  // Было +9 → +5
                } else {
                    activeRope = { targetX: pos.x, targetY: pos.y, maxLength: 380 };
                }
            }
        });

        canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            const pos = getEventPos(e.touches[0]);
            const dx = pos.x - ball.x;
            const dy = pos.y - ball.y;
            if (Math.sqrt(dx*dx + dy*dy) < ball.radius * 1.5) {
                if (!gameStarted) {
                    gameStarted = true;
                    const randomAngle = (Math.random() - 0.5) * Math.PI;
                    const startSpeed = 16;
                    ball.vx = Math.cos(randomAngle) * startSpeed;
                    ball.vy = Math.sin(randomAngle) * startSpeed + 5;
                } else {
                    activeRope = { targetX: pos.x, targetY: pos.y, maxLength: 380 };
                }
            }
        });

        canvas.addEventListener('mousemove', (e) => {
            if (activeRope) {
                const pos = getEventPos(e);
                activeRope.targetX = pos.x;
                activeRope.targetY = pos.y;
            }
        });

        canvas.addEventListener('touchmove', (e) => {
            e.preventDefault();
            if (activeRope) {
                const pos = getEventPos(e.touches[0]);
                activeRope.targetX = pos.x;
                activeRope.targetY = pos.y;
            }
        });

        canvas.addEventListener('mouseup', () => activeRope = null);
        canvas.addEventListener('touchend', () => activeRope = null);

        function update() {
            centerX = canvas.width * 0.5;
            centerY = canvas.height * 0.4;
            
            if (gameStarted) {
                ball.vy += gravity;
            }
            
            ball.x += ball.vx;
            ball.y += ball.vy;
            
            updateRopes();
            ball.vx *= damping;
            ball.vy *= damping;
            
            constrainToRing();
        }

        function drawBackground() {
            const gradient = ctx.createRadialGradient(
                canvas.width * 0.3, canvas.height * 0.2, 0,
                canvas.width * 0.3, canvas.height * 0.2, canvas.width
            );
            gradient.addColorStop(0, '#E3F2FD');
            gradient.addColorStop(0.5, '#87CEEB');
            gradient.addColorStop(1, '#98FB98');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, canvas.width, canvas.height);
        }

        function drawClouds() {
            ctx.save();
            ctx.fillStyle = 'rgba(255,255,255,0.8)';
            ctx.shadowColor = 'rgba(255,255,255,0.3)';
            ctx.shadowBlur = 10;
            
            const time = Date.now() * 0.001;
            const clouds = [
                {x: canvas.width * 0.2 + Math.sin(time) * 20, y: 80, size: 60},
                {x: canvas.width * 0.6 + Math.cos(time * 0.8) * 15, y: 120, size: 80},
                {x: canvas.width * 0.85 + Math.sin(time * 0.9) * 25, y: 60, size: 70}
            ];
            
            clouds.forEach(cloud => {
                ctx.beginPath();
                ctx.arc(cloud.x, cloud.y, cloud.size, 0, Math.PI * 2);
                ctx.arc(cloud.x + 30, cloud.y, cloud.size * 0.7, 0, Math.PI * 2);
                ctx.arc(cloud.x + 60, cloud.y, cloud.size, 0, Math.PI * 2);
                ctx.arc(cloud.x + 20, cloud.y - 20, cloud.size * 0.5, 0, Math.PI * 2);
                ctx.fill();
            });
            ctx.shadowBlur = 0;
            ctx.restore();
        }

        function drawRing() {
            ctx.save();
            ctx.translate(centerX, centerY);
            ctx.strokeStyle = '#333';
            ctx.lineWidth = 20;
            ctx.shadowColor = 'rgba(0,0,0,0.5)';
            ctx.shadowBlur = 30;
            ctx.beginPath();
            ctx.arc(0, 0, ringRadius, 0, Math.PI * 2);
            ctx.stroke();
            ctx.shadowBlur = 0;
            ctx.restore();
        }

        function drawRopes() {
            ctx.lineCap = 'round';
            ctx.shadowBlur = 15;
            
            if (activeRope) {
                ctx.strokeStyle = '#ff1493';
                ctx.shadowColor = '#ff69b4';
                ctx.lineWidth = 10;
                ctx.beginPath();
                ctx.moveTo(ball.x, ball.y);
                ctx.lineTo(activeRope.targetX, activeRope.targetY);
                ctx.stroke();
            }
            
            ropes.forEach(rope => {
                const alpha = rope.life / 150;
                ctx.strokeStyle = `rgba(138, 43, 226, ${alpha * 0.8})`;
                ctx.shadowColor = `rgba(147, 0, 217, ${alpha * 0.6})`;
                ctx.lineWidth = 8 * alpha;
                ctx.beginPath();
                ctx.moveTo(ball.x, ball.y);
                ctx.lineTo(rope.targetX, rope.targetY);
                ctx.stroke();
            });
            
            ctx.shadowBlur = 0;
        }

        function drawBall() {
            ctx.save();
            ctx.shadowColor = 'rgba(0,0,0,0.5)';
            ctx.shadowBlur = 25;
            ctx.fillStyle = ball.color;
            ctx.beginPath();
            ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
            ctx.fill();
            ctx.shadowBlur = 0;
            
            ctx.fillStyle = 'rgba(255,255,255,0.7)';
            ctx.beginPath();
            ctx.arc(ball.x - 10, ball.y - 10, 8, 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();
        }

        function gameLoop() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            drawBackground();
            drawClouds();
            update();
            drawRing();
            drawRopes();
            drawBall();
            requestAnimationFrame(gameLoop);
        }

        spawnBall();
        gameLoop();
    </script>
</body>
</html>
